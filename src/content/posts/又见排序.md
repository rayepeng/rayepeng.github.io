我第一遍写的代码， 有点问题
```
template <class T1, class T2>
void mysort(T1 *a, T1 *b, T2 c)
{
	int len = b - a;
	for (int i = 0; i < len; i++)
	{
		for (int j = i+1; j <= len; j++)
		{
			if (!c(a[i], a[j]))
			{
				T1 k;
				k = a[j];
				a[j] = a[i];
				a[i] = k;
			}
		}
	}
}
```

正确的模板类代码

```
template <class T1, class T2>
void mysort(
	T1* s, T1* e, T2(*h)(T1 a, T1 b)) {
	int gap = e - s;
	for (int i = 0; i < gap - 1; i++)
		for (int j = 0; j < gap - 1 - i; j++)
		{
			T1 swap;
			if (!(*h)(*(s + j), *(s + j + 1)))
			{
				swap = *(s + j);
				*(s + j) = *(s + j + 1);
				*(s + j + 1) = swap;
			}
		}
}
```
