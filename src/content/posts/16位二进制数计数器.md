要设计一个自动加载型的16位二进制计数器

```verilog
module CNT1058(CLK, RST, EN, LOAD, COUT, DOUT, DATA);
	input CLK, EN, RST, LOAD;
	input [15:0] DATA;
	output [15:0] DOUT;
	output COUT;

	reg[15:0] Q1;
	
	reg FULL;

	assign DOUT = Q1;

	always @(posedge CLK or posedge LOAD or negedge RST)
	begin
		if(!RST)  //异步复位信号
		begin
			Q1 <= 0;
			FULL <= 0;
		end
		else if(LOAD)  //异步加载信号
		begin
			Q1 <= DATA;
			FULL <= 1;
		end
		else if(EN)  //同步使能信号
		begin
			Q1 <= Q1 + 1;
			FULL <= 0;
		end

	end

	assign LOAD = (Q1 == 16'b0) ; //产生加载信号
	assign DOUT = Q1;           //数据
	assign COUT = FULL;         //进位信号

endmodule

```

我们需要完成几个操作
1. 同步计数， Q1 <= Q1 + 1
2. 异步加载数据 Q1 <= DATA
3. 异步清零 

同时计数的时候， 如果加到了最大值要回来， 产生一个控制信号 LOAD
之后输出数据

问题是仿真的时候

module CNT1058_tb();

	reg CLK, EN, RST, LOAD;

	reg [15:0] DATA;

	wire [15:0] DOUT;
	wire COUT;

	initial
	begin
		RST = 0;
		CLK = 0;

		LOAD = 1;

		EN = 0;
		DATA = 16'b1111111111111100;

		# 20
		LOAD = 1;
		EN = 1;
		RST = 1;

	end

	always 
	begin
	# 10
		CLK = ~CLK;
	end

	CNT1058 C1(
		.CLK(CLK),
		.RST(RST),
		.EN(EN),
		.LOAD(LOAD),
		.COUT(COUT),
		.DOUT(DOUT),
		.DATA(DATA)
		);

endmodule
